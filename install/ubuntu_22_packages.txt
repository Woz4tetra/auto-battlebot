# Required packages for auto-battlebot (Ubuntu 22.04)
# One package per line, lines starting with # are ignored
# TensorRT runtime (libnvinfer-dev) is installed in step 0; listed here so apt install runs after repo is added.

libopencv-dev
libnvinfer-dev

# clang-tidy (installed by install/install_llvm_toolchain.sh) auto-selects the
# newest GCC toolchain it finds under /usr/lib/gcc. On 22.04, gcc-12 is pulled in
# as a transitive dep, so clang picks it and looks for libstdc++ headers under
# /usr/include/c++/12. Without libstdc++-12-dev those headers are absent and
# every file fails lint with "'array' file not found". Install it so the
# toolchain clang selects always has matching C++ standard-library headers.
libstdc++-12-dev

python3.11@ppa:deadsnakes/ppa
python3.11-venv
python3.11-dev
